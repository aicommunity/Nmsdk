cmake_minimum_required(VERSION 3.16)
project(TimeLearnerDecisionAlgorithms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(QT_MAJOR 5 CACHE STRING "Qt major matching the declared worker toolchain")
find_package(Qt${QT_MAJOR} REQUIRED COMPONENTS Core)
find_package(GTest REQUIRED)
add_library(Nmsdk::Qt::Core ALIAS Qt${QT_MAJOR}::Core)
if(NOT NMSDK_TRAINING_ALGORITHM_SOURCE_ROOT)
    get_filename_component(NMSDK_TRAINING_ALGORITHM_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
endif()
set(NMSDK_TRAINING_ALGORITHM_RUNTIME "${CMAKE_BINARY_DIR}/runtime")
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../Libraries/TrainingAlgorithms.cmake")
