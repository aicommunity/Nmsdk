    write('feedback-auditor-regression-tests.json',{'tests_run':auditor.testsRun,'successful':auditor.wasSuccessful(),'failures':len(auditor.failures),'errors':len(auditor.errors),'skipped':len(auditor.skipped)})
    assert auditor.wasSuccessful() and auditor.testsRun==10 and not auditor.skipped
    passed = True
except BaseException as error:
    write('exception.json', {'reason': str(error), 'traceback': traceback.format_exc()})
finally:
    stop.set()
    if thread:
        thread.join(timeout=30)
    write('terminal.json', {'build_passed': buildPassed, 'native_gates_executed': (out / 'native12/native.xml').exists(), 'profile_graph_import_scoped_acceptance_passed': passed, 'scientific_verdict': None, 'whole_plan_complete': False})

            for p in (source / prefix).rglob('*'):
                if p.is_file() and '__pycache__' not in p.parts:
                    hashes[p.relative_to(source).as_posix()] = hashlib.sha256(p.read_bytes()).hexdigest()
    write('source-hashes.json', hashes)

        for name, raw in files.items():
            info = tarfile.TarInfo(name)
            info.size = len(raw)
            archive.addfile(info, io.BytesIO(raw))
    raw = (out / 'primary.tar.xz').read_bytes()
    write('bundle.json', {'kind': 'profile_graph_import_native24_reader27', 'root': root, 'build_passed': buildPassed, 'native_gates_executed': (out / 'native12/native.xml').exists(), 'members': {n: {'bytes': len(v), 'sha256': hashlib.sha256(v).hexdigest()} for n, v in files.items()}, 'archive_bytes': len(raw), 'archive_sha256': hashlib.sha256(raw).hexdigest(), 'scientific_verdict': None})
    print('CLOSED', passed, flush=True)
