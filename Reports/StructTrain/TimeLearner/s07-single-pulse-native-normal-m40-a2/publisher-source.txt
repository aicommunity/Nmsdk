def git(*args, data=None):
    return subprocess.check_output(['git', '-C', str(repo), *args], input=data).decode().strip()

def tree(base_tree, changes):
    entries = {}
    if base_tree:
        raw = subprocess.check_output(['git', '-C', str(repo), 'ls-tree', '-z', base_tree])
        for item in raw.split(b'\x00'):
            if not item:
                continue
            header, name = item.split(b'\t', 1)
            mode, kind, sha = header.decode().split()
            entries[name.decode()] = (mode, kind, sha)
    groups = {}
    for path, value in changes.items():
        head, sep, tail = path.partition('/')
        if sep:
            groups.setdefault(head, {})[tail] = value
        else:
            entries[head] = value
    for name, sub in groups.items():
        previous = entries.get(name)
        assert previous is None or previous[1] == 'tree'
        entries[name] = ('040000', 'tree', tree(previous[2] if previous else None, sub))
    content = b''.join(((entries[n][0].lstrip('0') + ' ' + n).encode() + b'\x00' + bytes.fromhex(entries[n][2]) for n in sorted(entries, key=lambda n: n + ('/' if entries[n][1] == 'tree' else ''))))
    return git('hash-object', '-w', '-t', 'tree', '--stdin', data=content)
